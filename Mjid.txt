// 1. Cibler l'élément (par exemple, un champ de saisie)
const inputElement = document.querySelector('input');

// 2. Créer l'événement pour la touche Entrée
const enterEvent = new KeyboardEvent('keydown', {
    key: 'Enter',
    code: 'Enter',
    keyCode: 13,
    which: 13,
    bubbles: true, // Permet à l'événement de remonter dans le DOM
    cancelable: true
});

// 3. Déclencher l'événement sur l'élément
inputElement.dispatchEvent(enterEvent);

//============================================================
// Soumettre le formulaire directement
document.querySelector('form').submit();

// Ou simuler le clic sur le bouton de validation
document.querySelector('#submit-button').click();

//===============================================
function addProduct(productData) {
    const table = document.getElementById('idTable4');
    
    // Create row
    const newRow = document.createElement('tr');
    
    // Create and append cells
    const cell1 = document.createElement('td');
    cell1.textContent = productData.name;
    newRow.appendChild(cell1);
    
    const cell2 = document.createElement('td');
    cell2.textContent = productData.price;
    newRow.appendChild(cell2);
    
    const cell3 = document.createElement('td');
    cell3.textContent = productData.quantity;
    newRow.appendChild(cell3);
    
    const cell4 = document.createElement('td');
    cell4.textContent = productData.category;
    newRow.appendChild(cell4);
    
    // Append row to table
    table.appendChild(newRow);
}

// Usage
addProduct({
    name: 'Laptop',
    price: '$999',
    quantity: '5',
    category: 'Electronics'
});
